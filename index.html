const TAB_INDUK = "DATA ANGGOTA"; const TAB_KKA = "KKA"; // 1. Ambil data nombor siri BIL seterusnya function getNextBil(sheet) { let lastRow = sheet.getLastRow(); let nextBil = 1; if (lastRow >= 2) { let currentLastBil = sheet.getRange(lastRow, 1).getValue(); if (!isNaN(currentLastBil) && currentLastBil !== "") { nextBil = Number(currentLastBil) + 1; } } return nextBil; } // 2. Proses MKA & KKA function doPost(e) { try { const req = JSON.parse(e.postData.contents); const ss = SpreadsheetApp.getActiveSpreadsheet(); // ========================================== // ALIRAN MKA: MASUK KEKUATAN // ========================================== if (req.action === "MKA") { let sheetInduk = ss.getSheetByName(TAB_INDUK); let sheetCwg = ss.getSheetByName(req.targetTab); if (!sheetInduk) { return ContentService.createTextOutput(JSON.stringify({ status: "error", message: "Tab DATA ANGGOTA tidak dijumpai!" })).setMimeType(ContentService.MimeType.JSON); } if (!sheetCwg) { return ContentService.createTextOutput(JSON.stringify({ status: "error", message: "Tab cawangan " + req.targetTab + " tidak dijumpai!" })).setMimeType(ContentService.MimeType.JSON); } // Format baris asas (Lajur B hingga AI) let commonFields = [ req.noTentera || "", // B: NO TEN req.pangkat || "", // C: PKT req.nama || "", // D: NAMA req.mka || "", // E: MKA req.tred || "", // F: TRED req.noKpAnggota || "", // G: NO KP (ANGGOTA) req.bangsa || "", // H: BANGSA req.agama || "", // I: AGAMA req.statusKahwin || "", // J: STATUS req.tarikhLahir || "", // K: TARIKH LAHIR req.umur || "", // L: UMUR (TAHUN) req.tmpt || "", // M: TMPT req.tpt || "", // N: TPT req.tatademmot || "", // O: TATADEMOT req.jumlahTahunKhidmat || "", // P: JUMLAH TAHUN PERKHIDMATAN req.unitInsuran || "", // Q: UNIT INSURAN req.noAhliInsuran || "", // R: NO AHLI INSURAN req.noTel || "", // S: NO TELEFON req.pangkatPangkuan || "", // T: PANGKAT PANGKUAN req.pangkatTetap || "", // U: PANGKAT TETAP req.daerahMengundi || "", // V: DAERAH MENGUNDI req.dun || "", // W: DUN req.parlimen || "", // X: PARLIMEN req.negeri || "", // Y: NEGERI req.namaWaris || "", // Z: NAMA WARIS req.hubunganWaris || "", // AA: HUBUNGAN req.pekerjaanWaris || "", // AB: PEKERJAAN req.noKpWaris || "", // AC: NO K/P (WARIS) req.alamatWaris || "", // AD: ALAMAT WARIS req.noTelWaris || "", // AE: NO TEL (WARIS) req.bilAnak || "", // AF: BIL ANAK req.emailAirforce || "", // AG: EMAIL AIRFORCE req.emailPersonal || "", // AH: EMAIL PERSONAL req.catatan || "" // AI: CATATAN ]; // 1. Masukkan ke tab induk (DATA ANGGOTA) let bilInduk = getNextBil(sheetInduk); sheetInduk.appendRow([bilInduk, ...commonFields]); // 2. Masukkan ke tab Cawangan / Skn (kecuali jika pilihan memang DATA ANGGOTA) if (req.targetTab !== TAB_INDUK) { let bilCwg = getNextBil(sheetCwg); sheetCwg.appendRow([bilCwg, ...commonFields]); } return ContentService.createTextOutput(JSON.stringify({ status: "success", message: "Berjaya didaftarkan ke DATA ANGGOTA & " + req.targetTab })).setMimeType(ContentService.MimeType.JSON); } // ========================================== // ALIRAN KKA: KELUAR KEKUATAN // ========================================== if (req.action === "KKA") { let sheetKKA = ss.getSheetByName(TAB_KKA); if (!sheetKKA) { return ContentService.createTextOutput(JSON.stringify({ status: "error", message: "Tab KKA tidak dijumpai!" })).setMimeType(ContentService.MimeType.JSON); } let noTen = String(req.noTentera).trim(); let allSheets = ss.getSheets(); let rowDataSalinan = null; let asalUnit = ""; // Cari rekod daripada cawangan/skuadron dahulu for (let s = 0; s < allSheets.length; s++) { let sName = allSheets[s].getName(); if (sName === TAB_KKA || sName === TAB_INDUK || sName.startsWith("SPR")) continue; let data = allSheets[s].getDataRange().getValues(); for (let r = 1; r < data.length; r++) { if (String(data[r][1]).trim() === noTen) { rowDataSalinan = data[r].slice(); asalUnit = sName; allSheets[s].deleteRow(r + 1); // Padam dari cawangan break; } } if (rowDataSalinan) break; } // Padam dari tab DATA ANGGOTA let sheetInduk = ss.getSheetByName(TAB_INDUK); if (sheetInduk) { let dataInduk = sheetInduk.getDataRange().getValues(); for (let r = 1; r < dataInduk.length; r++) { if (String(dataInduk[r][1]).trim() === noTen) { if (!rowDataSalinan) { rowDataSalinan = dataInduk[r].slice(); asalUnit = "DATA ANGGOTA"; } sheetInduk.deleteRow(r + 1); // Padam dari DATA ANGGOTA break; } } } if (rowDataSalinan) { let bilKKA = getNextBil(sheetKKA); rowDataSalinan[0] = bilKKA; // Kemaskini BIL KKA rowDataSalinan[34] = "KKA dari " + asalUnit + " [" + (req.tarikhKKA || new Date().toLocaleDateString('ms-MY')) + "]: " + (req.catatanKKA || "Bertukar Keluar"); sheetKKA.appendRow(rowDataSalinan); return ContentService.createTextOutput(JSON.stringify({ status: "success", message: "Rekod dipindahkan ke KKA dan dipadam daripada DATA ANGGOTA & " + asalUnit })).setMimeType(ContentService.MimeType.JSON); } else { return ContentService.createTextOutput(JSON.stringify({ status: "not_found", message: "Nombor Tentera " + noTen + " tidak dijumpai dalam mana-mana unit!" })).setMimeType(ContentService.MimeType.JSON); } } return ContentService.createTextOutput(JSON.stringify({ status: "invalid_action" })).setMimeType(ContentService.MimeType.JSON); } catch (err) { return ContentService.createTextOutput(JSON.stringify({ status: "error", message: err.toString() })).setMimeType(ContentService.MimeType.JSON); } }
